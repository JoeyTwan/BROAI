# 通义千问（Dashscope）API 配置示例
# 复制此文件内容到 .env 文件（如果使用Vite等构建工具）
# 或参考 README.md 中的配置方法

# API密钥（从 https://dashscope.aliyun.com 获取）
VITE_OPENAI_API_KEY=your_api_key_here

# API地址（通常不需要修改）
VITE_OPENAI_API_URL=https://dashscope.aliyuncs.com/compatible-mode/v1/chat/completions

# 模型名称（可选：qwen-plus, qwen-turbo, qwen-max等）
VITE_OPENAI_MODEL=qwen-plus
